           {

            }
            System.out.println();
          }

                System.out.print("*");
            }
            System.out.println();
        }
    }
}

pattern-10

class Solution {
    public void pattern10(int n) {
for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= i; j++) {
                System.out.print("*");
            }
            System.out.println();
        }

        // Decreasing
        for (int i = n - 1; i >= 1; i--) {
            for (int j = 1; j <= i; j++) {
                System.out.print("*");
            }
            System.out.println();
        }
    }
}
